Estimate a causal effect under informative follow-up

Fits stabilized inverse-probability-of-censoring weights, then estimates the structural nested-model parameter using a weighted treatment model.

dplyrglmIPCWg-estimation
estimate_psi <- function(d, outcome, censor, covariates) {
  denominator_formula <- reformulate(
    c("treatment", covariates), response = censor
  )

  denominator_fit <- glm(denominator_formula, data = d,
                         family = binomial())
  numerator_fit <- glm(reformulate("treatment", response = censor),
                       data = d, family = binomial())

  weighted <- d |>
    dplyr::mutate(
      p_denom = predict(denominator_fit, type = "response"),
      p_num   = predict(numerator_fit, type = "response"),
      sw      = p_num / p_denom
    ) |>
    dplyr::filter(.data[[censor]] == 0, is.finite(sw))

  treatment_fit <- glm(reformulate(covariates, response = "treatment"),
                       data = weighted, family = binomial(), weights = sw)
  p_treatment <- predict(treatment_fit, type = "response")
  y <- weighted[[outcome]]

  sum(weighted$sw * y * (weighted$treatment - p_treatment)) /
    sum(weighted$sw * weighted$treatment *
          (weighted$treatment - p_treatment))
}

R / Research data management

Standardize records before modeling

Illustrates an explicit preparation step: preserve a deterministic participant identifier, enforce one record per visit, flag duplicates, and retain a concise audit summary.

data cleaningvalidationaudit trail
prepare_longitudinal_file <- function(visits) {
  cleaned <- visits |>
    dplyr::mutate(
      participant_id = as.character(participant_id),
      visit_number   = as.integer(visit_number),
      visit_date     = as.Date(visit_date)
    ) |>
    dplyr::filter(!is.na(participant_id), !is.na(visit_number))

  duplicates <- cleaned |>
    dplyr::count(participant_id, visit_number) |>
    dplyr::filter(n > 1)

  if (nrow(duplicates) > 0) {
    stop("Resolve duplicate participant-visit records before analysis.")
  }

  list(
    analytic_data = cleaned |> dplyr::arrange(participant_id, visit_number),
    audit = list(rows = nrow(cleaned),
                 participants = dplyr::n_distinct(cleaned$participant_id),
                 duplicate_pairs = 0)
  )
}

Make eligibility and temporal rules explicit

A concise pattern for creating a reproducible cohort table from linked hospital records while retaining the derivation logic needed for audit and review.

build_cohort <- function(admissions, encounters, index_events) {
  admissions |>
    dplyr::inner_join(encounters, by = c("patient_id", "admission_id")) |>
    dplyr::inner_join(index_events, by = c("patient_id", "admission_id")) |>
    dplyr::mutate(
      age_at_admission = as.numeric(admit_date - birth_date) / 365.25,
      eligible_age     = age_at_admission >= 18,
      event_in_window  = event_time >= admit_date & event_time <= discharge_date,
      index_time       = pmax(event_time, admit_date)
    ) |>
    dplyr::filter(eligible_age, event_in_window) |>
    dplyr::group_by(patient_id, admission_id) |>
    dplyr::slice_min(index_time, n = 1, with_ties = FALSE) |>
    dplyr::ungroup()
}
